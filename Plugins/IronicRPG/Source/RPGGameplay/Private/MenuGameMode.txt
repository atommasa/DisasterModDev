#include "MenuGameMode.h"


void AMenuGameMode::InitGame(const FString& MapName, const FString& Options, FString& ErrorMessage)
{
	Super::InitGame(MapName, Options, ErrorMessage);

	UISubsystem = GetGameInstance()->GetSubsystem<UUISubsystem>();
	if (UISubsystem && MenuWidgetClass)
	{
		if (UMenuBase* NewMenu = Cast<UMenuBase>(UISubsystem->OpenUIByClass(MenuWidgetClass)))
		{
			MenuWidget = NewMenu;

			OnMenuInitialized();
		}
		else
		{
			UE_LOG(LogRPGGameMode, Warning, TEXT("Failed to create menu widget of class %s"), *MenuWidgetClass->GetName());
		}
	}
}

void AMenuGameMode::StartGameSession(const FString& SlotName)
{
	ULoadingScreenSubsystem* LoadingScreenSubsystem = ULoadingScreenSubsystem::Get(this);
	if (LoadingScreenSubsystem)
	{
		LoadingScreenSubsystem->StartLoadingScreen();
	}

	SaveSubsystem->LoadGame(SlotName);
}
